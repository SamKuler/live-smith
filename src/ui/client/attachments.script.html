window.LiveSmithFactories = window.LiveSmithFactories || {}; window.LiveSmithFactories.createAttachments = function createAttachments({ deleteAttachment, getState, showStatus, uploadAttachments, }) { const imageMediaTypes = new Set([ "image/png", "image/jpeg", "image/webp", ]); const documentMediaTypes = new Set([ "application/pdf", "application/vnd.openxmlformats-officedocument.wordprocessingml.document", "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet", "application/vnd.openxmlformats-officedocument.presentationml.presentation", ]); const extensionKinds = new Map([ ["png", "image"], ["jpg", "image"], ["jpeg", "image"], ["webp", "image"], ["pdf", "document"], ["docx", "document"], ["xlsx", "document"], ["pptx", "document"], ]); const macroEnabledExtensions = new Set([ "docm", "dotm", "xlsm", "xltm", "xlam", "pptm", "potm", "ppsm", "sldm", ]); const legacyOfficeExtensions = new Set([ "doc", "dot", "xls", "xlt", "ppt", "pot", "pps", ]); const maxImageAttachmentBytes = __MAX_IMAGE_ATTACHMENT_BYTES__; const maxDocumentAttachmentBytes = __MAX_DOCUMENT_ATTACHMENT_BYTES__; const maxPendingAttachmentCount = __MAX_PENDING_ATTACHMENT_COUNT__; const maxPendingImageBytes = __MAX_PENDING_IMAGE_ATTACHMENT_BYTES__; const maxPendingTotalBytes = __MAX_PENDING_TOTAL_ATTACHMENT_BYTES__; const supportedFileMessage = "only PNG, JPEG, WebP, PDF, DOCX, XLSX, and PPTX are supported"; let busy = false; let initialized = false; function pendingAttachments() { return Array.isArray(getState().pendingAttachments) ? getState().pendingAttachments : []; } function formatBytes(byteLength) { if (byteLength < 1024) return `${byteLength} B`; const kibibytes = byteLength / 1024; if (kibibytes < 1024) return `${formatUnit(kibibytes)} KiB`; return `${formatUnit(kibibytes / 1024)} MiB`; } function formatUnit(value) { return value >= 10 || Number.isInteger(value) ? String(Math.round(value)) : value.toFixed(1); } function attachmentTypeLabel(attachment) { if (attachment.mediaType === "image/png") return "PNG"; if (attachment.mediaType === "image/jpeg") return "JPEG"; if (attachment.mediaType === "image/webp") return "WebP"; if (attachment.mediaType === "application/pdf") return "PDF"; if (attachment.mediaType.includes("wordprocessingml")) return "DOCX"; if (attachment.mediaType.includes("spreadsheetml")) return "XLSX"; if (attachment.mediaType.includes("presentationml")) return "PPTX"; return attachment.kind === "image" ? "Image" : "Document"; } function renderPending() { syncAttachAvailability(); const strip = document.getElementById("pendingAttachments"); if (!strip) return; strip.replaceChildren(); const attachments = pendingAttachments(); strip.hidden = attachments.length === 0; strip.setAttribute("aria-busy", String(busy)); for (const attachment of attachments) { const chip = document.createElement("span"); chip.className = "attachment-chip"; chip.dataset.attachmentId = attachment.id; chip.setAttribute("role", "listitem"); const label = document.createElement("span"); label.className = "attachment-chip-label"; label.textContent = `${attachment.fileName} · ${ attachmentTypeLabel(attachment) } · ${formatBytes(attachment.byteLength)}`; label.title = attachment.fileName; const remove = document.createElement("button"); remove.className = "attachment-remove"; remove.type = "button"; remove.disabled = busy; remove.setAttribute("aria-label", `Remove attached file ${attachment.fileName}`); remove.textContent = "×"; remove.addEventListener("click", () => { void removePending(attachment.id); }); chip.append(label, remove); strip.append(chip); } } function setBusy(value) { busy = Boolean(value); syncAttachAvailability(); document.querySelector(".composer") ?.setAttribute("aria-busy", String(busy)); renderPending(); } function activeSavedRuntime() { const state = getState(); const runtime = state.runtimeProfile; return runtime?.profile?.id && runtime.profile.id === state.settings?.activeProfileId ? runtime : null; } function imageInputStatus() { const runtime = activeSavedRuntime(); const evidence = runtime?.inputCapabilityEvidence?.image; if ( evidence === "supported" && runtime?.capabilities?.inputs?.image === true ) return "supported"; return evidence === "unsupported" ? attachment.byteLength : 0), 0, ); const accepted = []; const failures = []; for (const file of files) { const classification = classifyFile(file); let reason = classification.reason || ""; if (!reason && classification.kind === "image" && imageInputStatus() !== "supported") { reason = imageCapabilityReason(); } else if (!reason && file.size === 0) { reason = "the file is empty"; } else if ( !reason && classification.kind === "image" && file.size > maxImageAttachmentBytes ) { reason = `the image is larger than ${formatBytes(maxImageAttachmentBytes)}`; } else if ( !reason && classification.kind === "document" && file.size > maxDocumentAttachmentBytes ) { reason = `the document is larger than ${formatBytes(maxDocumentAttachmentBytes)}`; } else if (!reason && count >= maxPendingAttachmentCount) { reason = `a Session can have at most ${maxPendingAttachmentCount} pending files`; } else if (!reason && totalBytes + file.size > maxPendingTotalBytes) { reason = `pending files would exceed ${formatBytes(maxPendingTotalBytes)}`; } else if ( !reason && classification.kind === "image" && imageBytes + file.size > maxPendingImageBytes ) { reason = `pending images would exceed ${formatBytes(maxPendingImageBytes)}`; } if (reason) { failures.push(`${file.name || "Unnamed file"}: ${reason}`); continue; } accepted.push(file); count += 1; totalBytes += file.size; if (classification.kind === "image") imageBytes += file.size; } return { accepted, failures }; } function failureMessage(failures) { if (!failures.length) return ""; return `Some files were not attached: ${failures.join("; ")}. "The active model does not support image input. Remove the attached images or activate an image-capable Profile." : "Image input support for the active model is unverified. Remove the attached images, discover model capabilities, or set an explicit image-input override.", true, ); return false; } } if ( attachments.some((attachment) => attachment.mediaType === "application/pdf") && !pdfInputSupported() ) { showStatus( "PDF attachments require verified PDF input support with OpenAI Responses or Anthropic Messages. Remove the attached PDFs or activate a compatible saved Profile.", true, ); return false; } return true; } function filesFromTransfer(transfer) { return transfer?.files ? [...transfer.files] : []; } function hasUploadableFile(files) { return selectUploadableFiles(files).accepted.length > 0; } function initialize() { if (initialized) return; initialized = true; const composer = document.querySelector(".composer"); const prompt = document.getElementById("prompt"); const attachButton = document.getElementById("attachFileButton"); const input = document.getElementById("attachmentInput"); attachButton?.addEventListener("click", () => input?.click()); input?.addEventListener("change", () => { const selected = filesFromTransfer(input); input.value = ""; void uploadFiles(selected); }); prompt?.addEventListener("paste", (event) => { const files = filesFromTransfer(event.clipboardData); if (!files.length) return; if (hasUploadableFile(files)) event.preventDefault(); void uploadFiles(files); }); composer?.addEventListener("dragover", (event) => { const files = filesFromTransfer(event.dataTransfer); if (!files.length) return; event.preventDefault(); composer.classList.add("drop-target"); }); composer?.addEventListener("dragleave", (event) => { if (!composer.contains(event.relatedTarget)) { composer.classList.remove("drop-target"); } }); composer?.addEventListener("drop", (event) => { const files = filesFromTransfer(event.dataTransfer); if (!files.length) return; event.preventDefault(); composer.classList.remove("drop-target"); void uploadFiles(files); }); syncAttachAvailability(); renderPending(); } return { initialize, removePending, renderPending, setBusy, uploadFiles, validateSend, }; };
